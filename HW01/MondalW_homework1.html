from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score
def eval_model(name, model, X_tr, y_tr, X_te, y_te):
    model.fit(X_tr, y_tr)
    pred_tr=model.predict(X_tr)
    pred_te=model.predict(X_te)

    final_est = model.named_steps["model"]
    coef = getattr(final_est, "coef_", None)
    intercept = getattr(final_est, "intercept_", None)
    
    return {
    "Model": name,
     "Train_RMSE": float(np.sqrt(mean_squared_error(y_tr, pred_tr))),
     "Test_RMSE": float(np.sqrt(mean_squared_error(y_te,pred_te))), 
      "Train_MSE": float(mean_absolute_error(y_tr,pred_tr)),
      "Test_MSE": float(mean_absolute_error(y_te, pred_te)),
      "Train_R2": float(r2_score(y_tr,pred_tr)),
       "Test_R2": float(r2_score(y_te,pred_te)),
        "coff": coef,
        "Feature": X_tr.columns,
        "inter":intercept
    }

import numpy as np
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score

def eval_model(name, model, X_tr, y_tr, X_te, y_te, zero_tol=1e-10):
    model.fit(X_tr, y_tr)

    pred_tr = model.predict(X_tr)
    pred_te = model.predict(X_te)

    
    final_est = model.named_steps["model"] 

    coef = np.ravel(final_est.coef_)
    intercept = float(np.ravel(final_est.intercept_)[0])

    return {
        "Model": name,
        "Train_RMSE": float(np.sqrt(mean_squared_error(y_tr, pred_tr))),
        "Test_RMSE": float(np.sqrt(mean_squared_error(y_te, pred_te))),
        "Train_MSE": float(mean_absolute_error(y_tr, pred_tr)),
        "Test_MSE": float(mean_absolute_error(y_te, pred_te)),
        "Train_R2": float(r2_score(y_tr, pred_tr)),
        "Test_R2": float(r2_score(y_te, pred_te)),

        # coefficient diagnostics
        "Intercept": intercept,
        "Num_Features": int(coef.size),
        "NonZero_Coeff": int(np.sum(np.abs(coef) > zero_tol)),
        "Killed_Coeff": int(np.sum(np.abs(coef) <= zero_tol)),
        "L2_Norm": float(np.linalg.norm(coef, 2)),
        "L1_Norm": float(np.linalg.norm(coef, 1)),
        "Mean_Abs_Coeff": float(np.mean(np.abs(coef))),
        "Max_Abs_Coeff": float(np.max(np.abs(coef))),
    }

Which method performs best in training data?¶

Based on the training data performance, the method that performs best is Ordinary Least Squares (OLS), although the difference compared to Ridge is extremely small.

Looking at the training metrics:

OLS has the lowest Train RMSE (0.720799) and a high Train R² (0.609603).

Ridge (α = 10) performs almost identically, with Train RMSE = 0.720808 and Train R² = 0.609593, differing only at the fourth decimal place.

Lasso (α = 0.01) performs slightly worse on training data, with a higher Train RMSE (0.724630) and lower Train R² (0.605442), which is expected because Lasso applies stronger regularization and shrinks coefficients.

Therefore, OLS performs best on the training data because it minimizes training error without any regularization constraints. Ridge performs nearly the same due to mild coefficient shrinkage, while Lasso sacrifices some training accuracy in exchange for better regularization and potential generalization benefits.

Which method performs best in test data?¶

Based on the test data performance, the method that performs best is Lasso regression with α = 0.01.

Looking at the test metrics:

Ridge (α = 10) performs very similarly but slightly worse, with Test RMSE = 0.733053 and Test R² = 0.596866.

Although the differences are small, Lasso provides the best generalization performance because its regularization reduces overfitting while retaining predictive power. Therefore, Lasso with α = 0.01 performs best on the test datase
